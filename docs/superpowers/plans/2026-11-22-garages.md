# SuiviAuto 1.1 — Garages et contacts : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** remplacer le texte libre « garage » des interventions par un carnet de garages (coordonnées, horaires, spécialités, note, avis) avec contacts internes, sans perte pour les bases et sauvegardes 1.0.

**Architecture :**
- Deux nouveaux modèles SwiftData (`Garage`, `GarageContact`) et une relation optionnelle `Intervention.garageRef`. La migration légère est automatique.
- Une conversion idempotente (`GarageMigrator`) transforme au lancement les anciens textes en fiches.
- La logique pure (`ContactLinks`, clés de nom) et les services (`GarageService`) sont testés.
- Les vues SwiftUI suivent les modèles existants : `Card`, `saveOrRollback`, `tracksSheetPresentation`.

**Tech Stack :** Swift (mode 5), SwiftUI, SwiftData, XCTest, XcodeGen, Xcode 27.2, macOS 27.

**Spec :** `docs/superpowers/specs/2026-11-22-garages-design.md`

```swift
import Foundation

/// Liens d'action à partir des coordonnées saisies. Retourne nil si la valeur est inutilisable.
enum ContactLinks {
    static func phoneURL(_ text: String) -> URL? {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        let digits = String(trimmed.unicodeScalars.filter { CharacterSet(charactersIn: "0123456789").contains($0) })
        guard !digits.isEmpty else { return nil }
        let prefix = trimmed.hasPrefix("+") ? "+" : ""
        return URL(string: "tel:\(prefix)\(digits)")
    }

    static func mailURL(_ text: String) -> URL? {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard trimmed.contains("@"), !trimmed.contains(where: \.isWhitespace) else { return nil }
        return URL(string: "mailto:\(trimmed)")
    }

    static func mapsURL(_ address: String) -> URL? {
        let oneLine = address.components(separatedBy: .newlines)
            .map { $0.trimmingCharacters(in: .whitespaces) }
            .filter { !$0.isEmpty }
            .joined(separator: ", ")
        guard !oneLine.isEmpty else { return nil }
        var components = URLComponents(string: "https://maps.apple.com/")!
        components.queryItems = [URLQueryItem(name: "q", value: oneLine)]
        return components.url
    }

    static func webURL(_ text: String) -> URL? {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty, !trimmed.contains(where: \.isWhitespace) else { return nil }
        return URL(string: trimmed.contains("://") ? trimmed : "https://\(trimmed)")
    }
}
```

```swift
import Foundation
import SwiftData

/// Convertit les noms de garage texte (1.0) en fiches Garage. Idempotent ; ne modifie jamais le texte d'origine.
enum GarageMigrator {
    struct Result: Equatable {
        let created: Int
        let linked: Int
    }

    @discardableResult
    static func run(in context: ModelContext) throws -> Result {
        let pending = try context.fetch(FetchDescriptor<Intervention>(sortBy: [SortDescriptor(\.date)]))
            .filter { $0.garageRef == nil && !$0.garageTextMigrated }
        guard !pending.isEmpty else { return Result(created: 0, linked: 0) }

        var byKey: [String: Garage] = [:]
        for garage in try context.fetch(FetchDescriptor<Garage>(sortBy: [SortDescriptor(\.createdAt)])) {
            let key = Garage.nameKey(garage.name)
            if byKey[key] == nil { byKey[key] = garage }
        }

        var created = 0
        var linked = 0
        for intervention in pending {
            intervention.garageTextMigrated = true
            let cleaned = Garage.cleanedName(intervention.garage)
            guard !cleaned.isEmpty else { continue }
            let key = Garage.nameKey(cleaned)
            let garage: Garage
            if let existing = byKey[key] {
                garage = existing
            } else {
                garage = Garage(name: cleaned)
                context.insert(garage)
                byKey[key] = garage
                created += 1
            }
            intervention.garageRef = garage
            linked += 1
        }
        try DeletionService.saveOrRollback(context)
        return Result(created: created, linked: linked)
    }
}
```

```swift
import Foundation
import SwiftData

enum GarageService {
    private static let locale = Locale(identifier: "fr_FR")

    /// Tri alphabétique français, insensible à la casse et aux accents.
    static func sorted(_ garages: [Garage]) -> [Garage] {
        garages.sorted {
            $0.name.compare($1.name, options: [.caseInsensitive, .diacriticInsensitive], range: nil, locale: locale)
                == .orderedAscending
        }
    }

    static func hasDuplicateName(_ name: String, among garages: [Garage], excluding: Garage?) -> Bool {
        let key = Garage.nameKey(name)
        guard !key.isEmpty else { return false }
        return garages.contains { $0 !== excluding && Garage.nameKey($0.name) == key }
    }

    /// Les interventions liées gardent le nom du garage en texte ; les contacts sont supprimés.
    static func delete(_ garage: Garage, in context: ModelContext) throws {
        for intervention in garage.interventions {
            intervention.garage = garage.name
            intervention.garageTextMigrated = true
            intervention.garageRef = nil
        }
        context.delete(garage)
        try DeletionService.saveOrRollback(context)
    }

    /// Toutes voitures confondues, de la plus récente à la plus ancienne.
    static func interventions(of garage: Garage) -> [Intervention] {
        garage.interventions.sorted { ($0.date, $0.mileage) > ($1.date, $1.mileage) }
    }

    /// Total des factures (devis exclus), en centimes.
    static func invoiceTotalCents(of garage: Garage) -> Int {
        garage.interventions
            .filter { $0.status == .facture }
            .compactMap(\.amountCents)
            .reduce(0, +)
    }
}
```

```swift
            @unknown default: break
            }
        }
    }
```
