# SuiviAuto 2.0 — Plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** échéances par date ou par km avec devis, carnet d'entretien avec opérations détaillées et export PDF/CSV, pièces jointes dès la saisie avec lecture automatique des factures, photos sans rognage, icône rafraîchie, sans perte de données 1.0 ni 1.1.

**Architecture :**
- La logique pure va dans `Logic/` et est testée : `DeadlineScheduler`, `MaintenanceOperation`, `InvoiceTextParser`, `Carnet`.
- Les services vont dans `Services/` : `InvoiceOCRService` (PDFKit + Vision), `CarnetPDFExporter` (CoreText), `DeadlineService`, `PendingAttachments`.
- Les modèles SwiftData ne reçoivent que des ajouts, ce qui permet une migration légère.
- Les vues suivent les conventions existantes.

**Tech Stack :** Swift (mode 5), SwiftUI, SwiftData, Vision, PDFKit, CoreText, XCTest, XcodeGen, Xcode 27.2, macOS 27.

**Spec :** `docs/superpowers/specs/2026-11-23-v2-design.md`

```swift
import Foundation

/// Catalogue des opérations d'entretien (ordre = ordre d'affichage et de stockage).
enum MaintenanceOperation: String, CaseIterable, Identifiable, Codable {
    case vidange, courroie, liquideFrein, freins, filtres, pneus, controleTechnique, autre

    var id: String { rawValue }

    var label: String {
        switch self {
        case .vidange: "Vidange"
        case .courroie: "Courroie de distribution / accessoires"
        case .liquideFrein: "Liquide de frein"
        case .freins: "Plaquettes / Disques"
        case .filtres: "Filtres"
        case .pneus: "Pneumatiques"
        case .controleTechnique: "Contrôle technique"
        case .autre: "Autre"
        }
    }

    var symbol: String {
        switch self {
        case .vidange: "drop.fill"
        case .courroie: "gearshape.2.fill"
        case .liquideFrein: "drop.triangle.fill"
        case .freins: "circle.circle"
        case .filtres: "line.3.horizontal.decrease.circle.fill"
        case .pneus: "circle.dashed"
        case .controleTechnique: "checkmark.seal.fill"
        case .autre: "ellipsis.circle.fill"
        }
    }

    /// Règle d'entretien par défaut correspondante (voir MaintenanceRule.defaults).
    var suggestedRuleKey: String? {
        switch self {
        case .vidange: "vidange"
        case .courroie: "courroie"
        case .liquideFrein: "liquide-frein"
        default: nil
        }
    }

    /// Mots-clés (déjà repliés : minuscules, sans accents).
    var keywords: [String] {
        switch self {
        case .vidange: ["vidange", "huile moteur"]
        case .courroie: ["courroie", "distribution", "galet", "pompe a eau"]
        case .liquideFrein: ["liquide de frein", "purge"]
        case .freins: ["plaquette", "disque"]
        case .filtres: ["filtre"]
        case .pneus: ["pneu", "pneumatique", "equilibrage", "parallelisme"]
        case .controleTechnique: ["controle technique"]
        case .autre: []
        }
    }

    static func fold(_ text: String) -> String {
        text.folding(options: [.caseInsensitive, .diacriticInsensitive], locale: Locale(identifier: "fr_FR"))
    }

    /// Opérations reconnues dans un texte, dans l'ordre du catalogue.
    static func detect(in text: String) -> [MaintenanceOperation] {
        let folded = fold(text)
        return allCases.filter { op in op.keywords.contains { folded.contains($0) } }
    }

    /// Codes séparés par des virgules -> opérations connues, dédoublonnées, ordre du catalogue.
    static func decode(_ raw: String) -> [MaintenanceOperation] {
        let codes = Set(raw.split(separator: ",").map { String($0).trimmingCharacters(in: .whitespaces) })
        return allCases.filter { codes.contains($0.rawValue) }
    }

    static func encode(_ operations: [MaintenanceOperation]) -> String {
        let set = Set(operations)
        return allCases.filter { set.contains($0) }.map(\.rawValue).joined(separator: ",")
    }
}
```

```swift
import Foundation

struct DeadlineDue: Equatable {
    let dueDate: Date?
    let dueKm: Int?
    let remainingKm: Int?
    let status: DueStatus
}

enum DeadlineScheduler {
    static func due(mode: DeadlineMode, dueDate: Date, targetMileage: Int?,
                    estimator: MileageEstimator, now: Date, calendar: Calendar = .current) -> DeadlineDue {
        switch mode {
        case .date:
            return DeadlineDue(dueDate: dueDate, dueKm: nil, remainingKm: nil,
                               status: DueStatusEvaluator.status(dueDate: dueDate, remainingKm: nil, now: now, calendar: calendar))
        case .km:
            guard let target = targetMileage else {
```

```swift
import Foundation

struct InvoiceExtraction: Equatable {
    var date: Date?
    var amountCents: Int?
    var mileage: Int?
    var operations: [MaintenanceOperation] = []

    var isEmpty: Bool { date == nil && amountCents == nil && mileage == nil && operations.isEmpty }
}

/// Extrait date, montant TTC, kilométrage et opérations d'un texte de facture ou de devis.
enum InvoiceTextParser {
    private static let months = ["janvier", "fevrier", "mars", "avril", "mai", "juin", "juillet",
                                 "aout", "septembre", "octobre", "novembre", "decembre"]
    private static let numericDate = try! Regex(#"(?<!\d)(\d{1,2})[/.\-](\d{1,2})[/.\-](\d{4}|\d{2})(?!\d)"#)
    private static let textDate = try! Regex(#"(?<!\d)(\d{1,2})(?:er)?\s+(janvier|fevrier|mars|avril|mai|juin|juillet|aout|septembre|octobre|novembre|decembre)\s+(\d{4})"#)
    private static let money = try! Regex(#"(?<![\d,.])(\d{1,3}(?:[   .]\d{3})*|\d+)[,.](\d{2})(?!\d)"#)
    private static let kmAfterNumber = try! Regex(#"(?<![\d,.])(\d{1,3}(?:[   .]\d{3})+|\d{3,7})\s*kms?\b"#)
    private static let kmAfterLabel = try! Regex(#"(?:kilometrage|km compteur|kms?)\s*:?\s*(\d{1,3}(?:[   .]\d{3})+|\d{3,7})(?![\d,])"#)

    static func parse(_ text: String, calendar: Calendar = .current) -> InvoiceExtraction {
        let lines = text.components(separatedBy: .newlines)
        let folded = lines.map { MaintenanceOperation.fold($0) }
        return InvoiceExtraction(date: findDate(folded, calendar: calendar),
                                 amountCents: findAmount(folded),
                                 mileage: findMileage(folded),
                                 operations: MaintenanceOperation.detect(in: text))
    }

    // MARK: - Date

    private static func findDate(_ lines: [String], calendar: Calendar) -> Date? {
        let preferred = lines.filter { $0.contains("date") || $0.contains("facture") || $0.contains("devis") }
        for line in preferred + lines {
            if let d = dates(in: line, calendar: calendar).first { return d }
        }
        return nil
    }

    private static func dates(in line: String, calendar: Calendar) -> [Date] {
        var result: [Date] = []
        for m in line.matches(of: numericDate) {
            guard let d = Int(m.output[1].substring ?? ""), let mo = Int(m.output[2].substring ?? ""),
                  var y = Int(m.output[3].substring ?? "") else { continue }
            if y < 100 { y += 2000 }
            if let date = validDate(y, mo, d, calendar) { result.append(date) }
        }
        for m in line.matches(of: textDate) {
            guard let d = Int(m.output[1].substring ?? ""),
                  let name = m.output[2].substring, let mo = months.firstIndex(of: String(name)),
                  let y = Int(m.output[3].substring ?? "") else { continue }
            if let date = validDate(y, mo + 1, d, calendar) { result.append(date) }
        }
        return result
    }

    private static func validDate(_ y: Int, _ m: Int, _ d: Int, _ calendar: Calendar) -> Date? {
        guard (1990...2100).contains(y), (1...12).contains(m), (1...31).contains(d) else { return nil }
        let comps = DateComponents(year: y, month: m, day: d)
        guard let date = calendar.date(from: comps) else { return nil }
        let back = calendar.dateComponents([.year, .month, .day], from: date)
        return (back.year == y && back.month == m && back.day == d) ? date : nil
    }

    // MARK: - Montant

    private static func findAmount(_ lines: [String]) -> Int? {
        let ttc = lines.filter { $0.contains("ttc") }
        let net = lines.filter { $0.contains("net a payer") }
        let total = lines.filter { $0.contains("total") && !$0.contains(" ht") && !$0.hasPrefix("ht") }
        for group in [ttc, net, total] {
            if let best = group.flatMap(amounts(in:)).max() { return best }
        }
        return nil
    }

    private static func amounts(in line: String) -> [Int] {
        line.matches(of: money).compactMap { m in
            guard let whole = m.output[1].substring, let cents = m.output[2].substring else { return nil }
            let digits = whole.filter(\.isNumber)
            guard digits.count <= 9, let euros = Int(digits), let c = Int(cents) else { return nil }
            return euros * 100 + c
        }
    }

    // MARK: - Kilométrage

    private static func findMileage(_ lines: [String]) -> Int? {
        for line in lines where line.contains("km") || line.contains("kilometrage") {
            for regex in [kmAfterLabel, kmAfterNumber] {
                for m in line.matches(of: regex) {
                    guard let raw = m.output[1].substring,
                          let value = Int(raw.filter(\.isNumber)),
                          (100...2_000_000).contains(value) else { continue }
                    return value
                }
            }
        }
        return nil
    }
}
```

```swift
import AppKit
import PDFKit
import Vision

/// Lit le texte d'une facture ou d'un devis (PDF ou image), sur le Mac et hors ligne, puis l'analyse.
enum InvoiceOCRService {
    enum OCRError: LocalizedError {
        case unreadable
        var errorDescription: String? { "Ce document ne peut pas être lu." }
    }

    static let maxPages = 3

    static func analyze(fileAt url: URL, calendar: Calendar = .current) async throws -> InvoiceExtraction {
        let text = try await Task.detached(priority: .userInitiated) { try recognizeText(at: url) }.value
        return InvoiceTextParser.parse(text, calendar: calendar)
    }

    static func recognizeText(at url: URL) throws -> String {
        switch url.pathExtension.lowercased() {
        case "pdf":
            guard let doc = PDFDocument(url: url) else { throw OCRError.unreadable }
            if let layer = doc.string, layer.trimmingCharacters(in: .whitespacesAndNewlines).count >= 20 {
                return layer
            }
            var parts: [String] = []
            for index in 0..<min(doc.pageCount, maxPages) {
                guard let page = doc.page(at: index), let image = render(page) else { continue }
                parts.append(try recognize(image))
            }
            return parts.joined(separator: "\n")
        case "jpg", "jpeg", "png", "heic":
            guard let source = CGImageSourceCreateWithURL(url as CFURL, nil),
                  let image = CGImageSourceCreateImageAtIndex(source, 0, nil) else { throw OCRError.unreadable }
            return try recognize(image)
        default:
            throw OCRError.unreadable
        }
    }

    private static func render(_ page: PDFPage, scale: CGFloat = 2.5) -> CGImage? {
        let bounds = page.bounds(for: .mediaBox)
        let width = Int(bounds.width * scale), height = Int(bounds.height * scale)
        guard width > 0, height > 0,
              let ctx = CGContext(data: nil, width: width, height: height, bitsPerComponent: 8, bytesPerRow: 0,
                                  space: CGColorSpaceCreateDeviceRGB(),
                                  bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
        ctx.setFillColor(CGColor(gray: 1, alpha: 1))
        ctx.fill(CGRect(x: 0, y: 0, width: width, height: height))
        ctx.scaleBy(x: scale, y: scale)
        page.draw(with: .mediaBox, to: ctx)
        return ctx.makeImage()
    }

    private static func recognize(_ image: CGImage) throws -> String {
        let request = VNRecognizeTextRequest()
        request.recognitionLevel = .accurate
        request.recognitionLanguages = ["fr-FR"]
        request.usesLanguageCorrection = true
        try VNImageRequestHandler(cgImage: image, options: [:]).perform([request])
        return (request.results ?? []).compactMap { $0.topCandidates(1).first?.string }.joined(separator: "\n")
    }
}
```

```swift
import Foundation

struct CarnetQuery: Equatable {
    var text = ""
    var operation: MaintenanceOperation?
    var year: Int?
    var garageID: UUID?
}

/// Carnet d'entretien : factures d'une voiture, filtrées, de la plus récente à la plus ancienne.
enum Carnet {
    static func entries(for car: Car, query: CarnetQuery, calendar: Calendar = .current) -> [Intervention] {
        let needle = query.text.trimmingCharacters(in: .whitespacesAndNewlines)
        return car.interventions
            .filter { $0.status == .facture }
            .filter { query.operation == nil || $0.operations.contains(query.operation!) }
            .filter { query.year == nil || calendar.component(.year, from: $0.date) == query.year }
            .filter { query.garageID == nil || $0.garageRef?.uuid == query.garageID }
            .filter { i in
                needle.isEmpty || [i.title, i.operationsSummary, i.garageDisplayName, i.notes, i.kind.label]
                    .contains { $0.localizedCaseInsensitiveContains(needle) }
            }
            .sorted { ($0.date, $0.mileage) > ($1.date, $1.mileage) }
    }

    static func totalCents(_ entries: [Intervention]) -> Int {
        entries.compactMap(\.amountCents).reduce(0, +)
    }

    static func years(for car: Car, calendar: Calendar = .current) -> [Int] {
        Set(car.interventions.filter { $0.status == .facture }.map { calendar.component(.year, from: $0.date) })
            .sorted(by: >)
    }

    static func garages(for car: Car) -> [Garage] {
        var seen = Set<UUID>()
        let garages = car.interventions.compactMap(\.garageRef).filter { seen.insert($0.uuid).inserted }
        return GarageService.sorted(garages)
    }

    /// Texte de la colonne « Opérations » : opérations, sinon titre, sinon catégorie.
    static func operationsText(_ i: Intervention) -> String {
        let ops = i.operationsSummary
        if !ops.isEmpty { return ops }
        return i.displayTitle
```

```swift
import AppKit
import CoreText

/// Carnet d'entretien imprimable (PDF A4, pagination automatique).
enum CarnetPDFExporter {
    private static let page = CGRect(x: 0, y: 0, width: 595, height: 842)
    private static let margin: CGFloat = 40

    static func render(car: Car, entries: [Intervention], estimatedMileage: Int?, calendar: Calendar = .current) -> Data {
        let body = attributedCarnet(car: car, entries: entries, estimatedMileage: estimatedMileage, calendar: calendar)
        let data = NSMutableData()
        var box = page
        guard let consumer = CGDataConsumer(data: data as CFMutableData),
              let ctx = CGContext(consumer: consumer, mediaBox: &box, nil) else { return Data() }
        let framesetter = CTFramesetterCreateWithAttributedString(body)
        let frameRect = page.insetBy(dx: margin, dy: margin)
        var location = 0
        var pageNumber = 1
        repeat {
            ctx.beginPDFPage(nil)
            let path = CGPath(rect: frameRect.insetBy(dx: 0, dy: 10).offsetBy(dx: 0, dy: 10), transform: nil)
            let frame = CTFramesetterCreateFrame(framesetter, CFRange(location: location, length: 0), path, nil)
            CTFrameDraw(frame, ctx)
            drawFooter("\(car.name) — page \(pageNumber)", in: ctx)
            let visible = CTFrameGetVisibleStringRange(frame)
            ctx.endPDFPage()
            guard visible.length > 0 else { break }
            location += visible.length
            pageNumber += 1
        } while location < body.length
        ctx.closePDF()
        return data as Data
    }

    private static func drawFooter(_ text: String, in ctx: CGContext) {
        let line = CTLineCreateWithAttributedString(NSAttributedString(string: text, attributes: [
            .font: NSFont.systemFont(ofSize: 8), .foregroundColor: NSColor.gray,
        ]))
        ctx.textPosition = CGPoint(x: margin, y: margin / 2)
        CTLineDraw(line, ctx)
    }

    private static func attributedCarnet(car: Car, entries: [Intervention], estimatedMileage: Int?,
                                         calendar: Calendar) -> NSAttributedString {
        let out = NSMutableAttributedString()
        let df = DateFormatter()
        df.calendar = calendar; df.timeZone = calendar.timeZone
        df.locale = Locale(identifier: "fr_FR"); df.dateFormat = "dd/MM/yyyy"

        func add(_ s: String, size: CGFloat, bold: Bool = false, color: NSColor = .black, tabs: [CGFloat] = []) {
            let style = NSMutableParagraphStyle()
            style.tabStops = tabs.map { NSTextTab(textAlignment: .left, location: $0) }
            style.paragraphSpacing = 4
            out.append(NSAttributedString(string: s + "\n", attributes: [
                .font: bold ? NSFont.boldSystemFont(ofSize: size) : NSFont.systemFont(ofSize: size),
                .foregroundColor: color, .paragraphStyle: style,
            ]))
        }

        add("Carnet d'entretien — \(car.name)", size: 18, bold: true)
        let identity = [car.brand, car.modelName, car.plate].filter { !$0.isEmpty }.joined(separator: " · ")
        if !identity.isEmpty { add(identity, size: 11, color: .darkGray) }
        var meta = "1re immatriculation : \(df.string(from: car.firstRegistrationDate))"
        if let km = estimatedMileage { meta += " · Kilométrage estimé : \(Fmt.km(km))" }
        meta += " · Édité le \(df.string(from: .now))"
        add(meta, size: 9, color: .darkGray)
        add("", size: 6)

        let tabs: [CGFloat] = [70, 140, 360]
        add("Date\tKm\tOpérations\tGarage", size: 9, bold: true, tabs: tabs)
        for i in entries {
            var line = "\(df.string(from: i.date))\t\(Fmt.km(i.mileage))\t\(Carnet.operationsText(i))\t\(i.garageDisplayName)"
            if let cents = i.amountCents { line += " — \(Money.format(cents: cents))" }
            add(line, size: 9, tabs: tabs)
        }
        add("", size: 6)
        add("\(entries.count) intervention(s) · Total : \(Money.format(cents: Carnet.totalCents(entries)))", size: 10, bold: true)
        add("SuiviAuto — créé par Guillaume LECOMTE", size: 8, color: .gray)
        return out
    }
}
```

```swift
import Foundation
import Observation
import SwiftData

/// Pièces jointes ajoutées dans un formulaire avant son enregistrement.
/// Les fichiers sont copiés tout de suite ; `discard()` les supprime si le formulaire est annulé.
@Observable
final class PendingAttachments {
    private(set) var documents: [StoredDocument] = []
    private let store: DocumentStore

    init(store: DocumentStore = .shared) { self.store = store }

    /// Copie les fichiers ; renvoie les messages d'erreur des fichiers refusés.
    @discardableResult
    func add(urls: [URL]) -> [String] {
        var rejected: [String] = []
        for url in urls {
            let access = url.startAccessingSecurityScopedResource()
            defer { if access { url.stopAccessingSecurityScopedResource() } }
            do { documents.append(try store.importFile(at: url)) }
            catch { rejected.append(error.localizedDescription) }
        }
        return rejected
    }

    func url(for document: StoredDocument) -> URL { store.url(for: document) }

    func remove(_ document: StoredDocument) {
        documents.removeAll { $0 === document }
        store.delete(fileName: document.fileName)
    }

    func attach(to intervention: Intervention, in context: ModelContext) {
        for doc in documents { context.insert(doc); doc.intervention = intervention }
        documents = []
    }

    func attach(to deadline: Deadline, in context: ModelContext) {
        for doc in documents { context.insert(doc); doc.deadline = deadline }
        documents = []
    }

    func discard() {
        documents.forEach { store.delete(fileName: $0.fileName) }
        documents = []
    }
}
```
